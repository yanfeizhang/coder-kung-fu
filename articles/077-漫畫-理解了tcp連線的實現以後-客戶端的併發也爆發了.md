    unsigned int hash = inet_ehashfn(net, daddr, hnum, saddr, sport);
    unsigned int slot = hash & hashinfo->ehash_mask;
    struct inet_ehash_bucket *head = &hashinfo->ehash[slot];

begin:
    //遍歷連結串列，逐個對比直到找到
    sk_nulls_for_each_rcu(sk, node, &head->chain) {

            continue;
        if (likely(INET_MATCH(sk, net, acookie,

            if (unlikely(!atomic_inc_not_zero(&sk->sk_refcnt)))
                goto begintw;
            if (unlikely(!INET_MATCH(sk, net, acookie,
                         saddr, daddr, ports, dif))) {
                sock_put(sk);
                goto begin;
            }
            goto out;
        }
    }
}

## define `INET_MATCH`(__sk, __net, __cookie, __saddr, __daddr, __ports, __dif) \
((`inet_sk`(__sk)->`inet_portpair` == (__ports))    &&        \
(`inet_sk`(__sk)->`inet_daddr`    == (__saddr))    &&        \
(`inet_sk`(__sk)->`inet_rcv_saddr`    == (__daddr))    &&        \
(!(__sk)->`sk_bound_dev_if`    ||                \
((__sk)->`sk_bound_dev_if` == (__dif)))     &&        \
`net_eq`(`sock_net`(__sk), (__net)))

「在   `INET_MATCH` 中將網路封包 tcp header 中的__saddr、\_\_daddr、\_\_ports 和 Linux 中的 `socket` 中 `inet_portpair`、 `inet_daddr`、  `inet_rcv_saddr` 進行對比。如果匹配 `socket` 就找到了。當然除了 ip 和埠， `INET_MATCH` 還比較了其他一些東西，所以 TCP 還有五元組、七元組之類的說法。」

## 總結

用戶端每建立一個連線就要消耗一個埠，所以很多同學當看到用戶端機器上連線數一旦超過 3W、5W 就緊張的不行，總覺得機器要出問題了。

這篇文章第一版很早就寫出來了，不過飛哥又打磨了有一週之多。在文中我們展示了一下 TCP   `socket` 的少量核心程式碼。透過原始碼來看：

TCP 連線就是在客戶機、伺服器上的一對的 `socket`。它們都在各自核心物件上記錄了雙方的 ip 對、連接埠配對（也就是我們常說的四元組），透過這個在通訊時找到對方。

TCP 連線傳送方在傳送網路封包的時候，會把這份資訊複製到 IP Header 上。網路封包帶著這份信物穿過網際網路 (Network)，到達目的伺服器。目的伺服器核心會按照 IP 包 header 中攜帶的信物（四元組）去匹配找到正確的   `socket`(連線)。

在這個過程裡我們可以看到，用戶端的埠只是這個四元組裡的一元而已。哪怕兩條連線用的是同一個埠號，只要用戶端 ip 不一樣，或者是伺服器不一樣都不影響核心正確尋找到對應的連線，而不會串線！

所以在用戶端增加 TCP 最大併發能力有兩個方法。第一個辦法，為用戶端設定多個 ip。第二個辦法，連線多個不同的 server。

> 不過這兩個辦法最好不要混用。因為使用多 IP 時，用戶端需要 `bind`。一旦 `bind` 之後，核心建立連線的時候就不會選擇用過的埠了。 `bind` 函式會改變核心選擇埠的策略~~

最後我們親手實驗證明了用戶端也可以突破百萬的併發量級。相信讀過此文的你，以後再也不用再懼怕 65535 這個數字了。
